  return TRANSLATIONS[currentLang]?.[key] ?? TRANSLATIONS.en[key] ?? key;
}

// Apply i18n text to static HTML elements
function applyStaticTranslations() {
  // Hero header
  const h1 = document.querySelector('.hero-title h1');
  if (h1) h1.textContent = i18n('header.title');
  const heroCta = document.querySelector('.hero-cta');
  if (heroCta) heroCta.textContent = i18n('header.cta');

  // Filter hint
  const hint = document.querySelector('.filter-hint');
  if (hint) hint.textContent = i18n('filter.hint');

  // Filter group labels (direct <label> children of .filter-group)
  const filterLabelMap = [
    ['ageFilter', 'filter.age_label'],
    ['sexFilter', 'filter.sex_label'],
    ['searchInput', 'filter.color_label'],
    ['energyLevel', 'filter.energy_label'],
    ['specialNeeds', 'filter.special_needs_label'],
    ['goodWithKids', 'filter.good_with_kids_label'],
    ['goodWithDogs', 'filter.good_with_dogs_label'],
    ['goodWithCats', 'filter.good_with_cats_label'],
  ];
  filterLabelMap.forEach(([filterName, key]) => {
    // Find the filter-group containing this filter's inputs
    const input = filterName === 'searchInput'
      ? document.getElementById('searchInput')
      : document.querySelector(`input[name="${filterName}"]`);
    if (!input) return;
    const group = input.closest('.filter-group');
    if (!group) return;
    const label = group.querySelector(':scope > label');
    if (label) label.textContent = i18n(key);
  });

  // Color placeholder
  const colorInput = document.getElementById('searchInput');
  if (colorInput) colorInput.placeholder = i18n('filter.color_placeholder');

  // Checkbox labels — map input name+value to i18n key
  const cbMap = {
    'ageFilter:young': 'filter.age_young',
    'ageFilter:adult': 'filter.age_adult',
    'ageFilter:senior': 'filter.age_senior',
    'sexFilter:male': 'filter.sex_male',
    'sexFilter:female': 'filter.sex_female',
    'energyLevel:unknown': 'filter.energy_unknown',
    'energyLevel:low': 'filter.energy_low',
    'energyLevel:medium': 'filter.energy_med',
    'energyLevel:high': 'filter.energy_high',
    'specialNeeds:yes': 'filter.special_needs_yes',
    'specialNeeds:no': 'filter.special_needs_no',
    'goodWithKids:unknown': 'filter.compat_unknown',
    'goodWithKids:yes': 'filter.compat_yes',
    'goodWithKids:somewhat': 'filter.compat_some',
    'goodWithKids:no': 'filter.compat_no',
    'goodWithDogs:unknown': 'filter.compat_unknown',
    'goodWithDogs:yes': 'filter.compat_yes',
    'goodWithDogs:somewhat': 'filter.compat_some',
    'goodWithDogs:no': 'filter.compat_no',
    'goodWithCats:unknown': 'filter.compat_unknown',
    'goodWithCats:yes': 'filter.compat_yes',
    'goodWithCats:somewhat': 'filter.compat_some',
    'goodWithCats:no': 'filter.compat_no',
  };
  Object.entries(cbMap).forEach(([nameVal, key]) => {
    const [name, value] = nameVal.split(':');
    const input = document.querySelector(`input[name="${name}"][value="${value}"]`);
    if (!input) return;
    const label = input.closest('label');
    if (!label) return;
    // Replace text node after the input
    const textNode = Array.from(label.childNodes).find(n => n.nodeType === 3 && n.textContent.trim());
    if (textNode) textNode.textContent = ' ' + i18n(key);
  });

  // Species tabs
  const speciesTabMap = { dog: 'species.dogs', cat: 'species.cats', small: 'species.smalls' };
  document.querySelectorAll('.species-tab').forEach(tab => {
    const species = tab.dataset.species;
    if (species && speciesTabMap[species]) {
      tab.textContent = i18n(speciesTabMap[species]);
    }

  const adoptLabel = document.querySelector('.adoption-links-label');

    ['modalGender', 'modal.sex_label'],

    if (!valueEl) return;
    const detailItem = valueEl.closest('.detail-item');
    if (!detailItem) return;
    const labelEl = detailItem.querySelector('.detail-label');
    if (labelEl) labelEl.textContent = i18n(key);
  });

  // Modal about heading
  const aboutH4 = document.querySelector('.modal-section h4');
  if (aboutH4) aboutH4.textContent = i18n('modal.about_heading');

  // Modal adopt CTA
  const ctaP = document.querySelector('.adopt-cta');

function updateHtmlLangAttribute() {
  document.documentElement.lang = currentLang;
}

function updatePageTitle() {
  document.title = i18n('page.title');
}

function updateToggleActiveState() {
  const btnEn = document.getElementById('langBtnEn');
  const btnEs = document.getElementById('langBtnEs');
  if (btnEn) btnEn.classList.toggle('active', currentLang === 'en');
  if (btnEs) btnEs.classList.toggle('active', currentLang === 'es');
}

function updateResultsCount(count) {
  const speciesKeys = { dog: 'results.species_dogs', cat: 'results.species_cats', small: 'results.species_small' };
  const species = i18n(speciesKeys[currentSpecies] || 'results.species_pets');
  const template = i18n('results.template');
  const el = document.querySelector('.results-count');
  if (el) el.innerHTML = template.replace('{count}', count).replace('{species}', species);
}

function updateSearcherCtaHref() {
  const cta = document.querySelector('a.hero-cta');
  if (!cta) return;
  const url = new URL(cta.href);
  if (currentLang === 'es') {
    url.searchParams.set('lang', 'es');
  } else {
    url.searchParams.delete('lang');
  }
  cta.href = url.toString();
}

async function init() {
  // Language detection from URL
  const urlParams = new URLSearchParams(window.location.search);
  const langParam = urlParams.get('lang');
  currentLang = (langParam === 'es') ? 'es' : 'en';

  // Apply static text translations + toggle state
  applyStaticTranslations();
  updateToggleActiveState();
  updateSearcherCtaHref();
  updateHtmlLangAttribute();
  updatePageTitle();

  // Language toggle click handler
  document.querySelectorAll('.lang-btn').forEach(btn => {
    btn.addEventListener('click', () => {
      const newLang = btn.dataset.lang;
      if (newLang === currentLang) return;

      currentLang = newLang;

      // Update URL without page reload
      const url = new URL(window.location.href);
      if (newLang === 'es') {
        url.searchParams.set('lang', 'es');
      } else {
        url.searchParams.delete('lang');
      }
      history.replaceState({}, '', url);

      // Re-render everything
      applyStaticTranslations();
      updateToggleActiveState();
      updateSearcherCtaHref();
      updateHtmlLangAttribute();
      updatePageTitle();
      applyFilters(); // re-renders card grid + empty state with new language
    });
  });

  // Event listeners for filters
  elements.searchInput.addEventListener('input', debounce(applyFilters, 300));
  
  // All filters are now checkboxes (multi-select)
  const checkboxFilters = [
    'ageFilter', 'sexFilter', 'energyLevel', 
    'goodWithCats', 'goodWithDogs', 'goodWithKids', 'specialNeeds'
  ];
  
  checkboxFilters.forEach(filterName => {
    document.querySelectorAll(`input[name="${filterName}"]`).forEach(cb => {
      cb.addEventListener('change', () => applyFilters());
    });
  });
  
  // Close modal on background click
  elements.modal.addEventListener('click', (e) => {
    if (e.target === elements.modal) closeModal();
  });
  
  // Keyboard: Escape closes, arrows navigate gallery
  document.addEventListener('keydown', (e) => {
    // Lightbox takes priority over modal
    const lightbox = document.querySelector('.photo-lightbox');
    if (lightbox) {
      if (e.key === 'Escape') { lightbox.remove(); return; }
      if (e.key === 'ArrowLeft') { e.preventDefault(); lightboxNav(-1); return; }
      if (e.key === 'ArrowRight') { e.preventDefault(); lightboxNav(1); return; }
      return;
    }
    if (!elements.modal.classList.contains('active')) return;
    if (e.key === 'Escape') closeModal();
  });
  
  // Load animals
  await loadAnimals();
}

// Switch species tab
function switchSpecies(species) {
  currentSpecies = species;
  elements.speciesFilter.value = species;
  
  // Update tab UI
  document.querySelectorAll('.species-tab').forEach(tab => {
    tab.classList.toggle('active', tab.dataset.species === species);
  });
  
  // Clear filters and reapply (updateResultsCount handles species label via template)
  clearFilters();
}


// Get placeholder SVG based on species
function getPlaceholder(species) {
  const s = (species || '').toLowerCase();
  if (s === 'cat') return 'data:image/svg+xml,' + encodeURIComponent(CAT_PLACEHOLDER);
  if (s === 'dog') return 'data:image/svg+xml,' + encodeURIComponent(DOG_PLACEHOLDER);
  return 'data:image/svg+xml,' + encodeURIComponent(SMALL_PLACEHOLDER);
}

async function loadAnimals() {
  try {
    const response = await fetch(`${API_BASE}/animals`);
    const result = await response.json();
    
    if (result.success && result.data) {
      // Store all animals - filtering happens in applyFilters
      allAnimals = result.data;
      applyFilters();
    } else {
      showError(i18n('error.load_failed'));
    }
  } catch (error) {
    console.error('Load error:', error);
    showError(i18n('error.connection'));
  } finally {
    elements.loadingState.style.display = 'none';
  }
}

// Check if an animal matches the current species tab
function matchesSpecies(animal) {
  const species = (animal.species || '').toLowerCase();
  
  if (currentSpecies === 'dog') {
    return species === 'dog';
  } else if (currentSpecies === 'cat') {
    return species === 'cat';
  } else if (currentSpecies === 'small') {
    // Match any small animal species
    return SMALL_SPECIES.some(s => species.includes(s));
  }
  return false;
}

// Truncate age: "8 years 9 months" → "8 yrs", "6 months" → "6 mos"
function truncateAgeToYears(ageText) {
  if (!ageText) return '';
  const yearsMatch = ageText.match(/^(\d+)\s*years?/i);
  if (yearsMatch) {
    const years = parseInt(yearsMatch[1], 10);
    return `${years} ${years === 1 ? i18n('card.age_yr') : i18n('card.age_yrs')}`;
  }
  const monthsMatch = ageText.match(/^(\d+)\s*months?/i);
  if (monthsMatch) {
    const months = parseInt(monthsMatch[1], 10);
    return `${months} ${months === 1 ? i18n('card.age_mo') : i18n('card.age_mos')}`;
  }
  return ageText;
}

// Gender: Male → M, Female → F
function abbreviateGender(sex) {
  const s = (sex || '').trim().toLowerCase();
  if (s === 'male') return i18n('card.sex_male');
  if (s === 'female') return i18n('card.sex_female');
  return i18n('card.sex_unknown');
}

// Energy: Low / Med / High / Unk
function abbreviateEnergy(matchValue) {
  const e = (matchValue || 'unknown').toLowerCase();
  if (e === 'low') return i18n('card.energy_low');
  if (e === 'medium') return i18n('card.energy_med');
  if (e === 'high') return i18n('card.energy_high');
  return i18n('card.energy_unknown');
}

// Compact symbol for compatibility _match enum values
function getCompatSymbol(matchValue) {
  switch ((matchValue || 'unknown').toLowerCase()) {
    case 'yes': return i18n('card.compat_yes');
    case 'somewhat': return i18n('card.compat_some');
    case 'no': return i18n('card.compat_no');
    case 'unknown':
    default: return i18n('card.compat_unknown');
  }
}

// "Kids ✓, Dogs ✓, Cats ?" for front card
function formatCompatValue(animal) {
  const bn = animal.behaviorNotes;
  const kids = getCompatSymbol(bn?.goodWithKids_match);
  const dogs = getCompatSymbol(bn?.goodWithDogs_match);
  const cats = getCompatSymbol(bn?.goodWithCats_match);
  return `${i18n('card.compat_kids')} ${kids}, ${i18n('card.compat_dogs')} ${dogs}, ${i18n('card.compat_cats')} ${cats}`;
}

// Energy label for front card
function formatEnergyValue(animal) {
  const energy = (animal.behaviorNotes?.energyLevel_match || 'unknown').toLowerCase();
  if (energy === 'unknown') return i18n('card.energy_unknown');
  if (energy === 'low') return i18n('card.energy_low');
  if (energy === 'medium') return i18n('card.energy_med');
  if (energy === 'high') return i18n('card.energy_high');
  return energy.charAt(0).toUpperCase() + energy.slice(1);
}

// Special needs Yes/No for front card
function formatSpecialNeedsValue(animal) {
  const sn = (animal.behaviorNotes?.specialNeeds || '').trim().toLowerCase();
  const hasNeeds = sn && sn !== 'none' && sn !== 'n/a' && sn !== 'no' && !sn.includes('no special') && sn !== 'not specified' && sn !== '-';
  return hasNeeds ? i18n('card.special_needs_yes') : i18n('card.special_needs_no');
}



function getSelectedCheckboxValues(name) {
  return Array.from(document.querySelectorAll(`input[name="${name}"]:checked:not(:disabled)`))
    .map(cb => cb.value.toLowerCase());
}

// Check if any filter checkbox is checked or color input has text
function hasActiveFilters() {
  const allFilterNames = ['ageFilter', 'sexFilter', 'energyLevel', 'goodWithCats', 'goodWithDogs', 'goodWithKids', 'specialNeeds'];
  for (const filterName of allFilterNames) {
    if (getSelectedCheckboxValues(filterName).length > 0) return true;
  }
  if (elements.searchInput && elements.searchInput.value.trim()) return true;
  return false;
}

function calculateMatchScore(animal) {
  let score = 50; // Base score
  
  // Age filter match — no defaults, any selection is user-narrowing
  const selectedAges = getSelectedCheckboxValues('ageFilter');
  if (selectedAges.length > 0) {
    const years = animal.ageInYears;
    let ageMatch = false;
    if (selectedAges.includes('young') && years < 2) ageMatch = true;
    if (selectedAges.includes('adult') && years >= 2 && years < 7) ageMatch = true;
    if (selectedAges.includes('senior') && years >= 7) ageMatch = true;
    if (ageMatch) score += 20;
  }
  
  // Gender filter match — no defaults, any selection is user-narrowing
  const selectedSex = getSelectedCheckboxValues('sexFilter');
  if (selectedSex.length > 0 && selectedSex.some(s => animal.sex.toLowerCase().startsWith(s))) {
    score += 10;
  }
  
  // Compatibility/energy filters — only score when user has narrowed beyond defaults
  const compatFilters = [
    { name: 'energyLevel', field: 'energyLevel_match', points: 15 },
    { name: 'goodWithCats', field: 'goodWithCats_match', points: 10 },
    { name: 'goodWithDogs', field: 'goodWithDogs_match', points: 10 },
    { name: 'goodWithKids', field: 'goodWithKids_match', points: 10 },
  ];
  
  for (const { name, field, points } of compatFilters) {
    const selected = getSelectedCheckboxValues(name);
    if (selected.length === 0) continue; // No boxes checked — no score impact
    
    const animalValue = animal.behaviorNotes?.[field];
    const isUnknown = !animalValue || animalValue === 'unknown';
    if (isUnknown) {
      if (selected.includes('unknown')) score += Math.round(points * 0.5); // Partial credit
    } else {
      if (selected.includes(animalValue)) score += points;
    }
  }
  
  // Color search match (check both ShelterManager color and behavior notes color)
  const search = elements.searchInput.value.toLowerCase().trim();
  if (search) {
    const smColor = (animal.color || '').toLowerCase();
    const bnColor = (animal.behaviorNotes?.color || '').toLowerCase();
    if (smColor.includes(search) || bnColor.includes(search)) {
      score += 15;
    }
  }
  
  // Bonus for having a description
  if (animal.description && animal.description.length > 50) {
    score += 5;
  }
  
  // Bonus for having behavior notes
  if (animal.behaviorNotes) {
    score += 5;
  }
  
  return Math.min(100, score);
}

function applyFilters() {
  const search = elements.searchInput.value.toLowerCase().trim();
  
  // Get all checkbox filter values
  const selectedAges = getSelectedCheckboxValues('ageFilter');
  const selectedSex = getSelectedCheckboxValues('sexFilter');
  const selectedEnergies = getSelectedCheckboxValues('energyLevel');
  const selectedGoodWithCats = getSelectedCheckboxValues('goodWithCats');
  const selectedGoodWithDogs = getSelectedCheckboxValues('goodWithDogs');
  const selectedGoodWithKids = getSelectedCheckboxValues('goodWithKids');
  const selectedSpecialNeeds = getSelectedCheckboxValues('specialNeeds');
  
  filteredAnimals = allAnimals.filter(animal => {
    // Species filter based on current tab
    if (!matchesSpecies(animal)) {
      return false;
    }
    
    // Age filter (multi-select checkboxes)
    if (selectedAges.length > 0) {
      const years = animal.ageInYears;
      let ageMatch = false;
      if (selectedAges.includes('young') && years < 2) ageMatch = true;
      if (selectedAges.includes('adult') && years >= 2 && years < 7) ageMatch = true;
      if (selectedAges.includes('senior') && years >= 7) ageMatch = true;
      if (!ageMatch) return false;
    }
    
    // Sex filter (multi-select checkboxes)
    if (selectedSex.length > 0) {
      const animalSex = animal.sex.toLowerCase().trim();
      const sexMatch = selectedSex.some(s => animalSex.startsWith(s));
      if (!sexMatch) return false;
    }
    
    // Energy level filter - use _match field only, no text fallback
    // "unknown" option includes animals with unknown/empty/missing _match
    if (selectedEnergies.length > 0) {
      const energyMatch = animal.behaviorNotes?.energyLevel_match;
      const isUnknown = !energyMatch || energyMatch === 'unknown';
      if (isUnknown) {
        if (!selectedEnergies.includes('unknown')) return false;
      } else {
        if (!selectedEnergies.includes(energyMatch)) return false;
      }
    }
    
    // Good with cats filter - use _match field only, no text fallback
    // "unknown" option includes animals with unknown/empty/missing _match
    if (selectedGoodWithCats.length > 0) {
      const catsMatch = animal.behaviorNotes?.goodWithCats_match;
      const isUnknown = !catsMatch || catsMatch === 'unknown';
      if (isUnknown) {
        if (!selectedGoodWithCats.includes('unknown')) return false;
      } else {
        if (!selectedGoodWithCats.includes(catsMatch)) return false;
      }
    }
    
    // Good with dogs filter - use _match field only, no text fallback
    // "unknown" option includes animals with unknown/empty/missing _match
    if (selectedGoodWithDogs.length > 0) {
      const dogsMatch = animal.behaviorNotes?.goodWithDogs_match;
      const isUnknown = !dogsMatch || dogsMatch === 'unknown';
      if (isUnknown) {
        if (!selectedGoodWithDogs.includes('unknown')) return false;
      } else {
        if (!selectedGoodWithDogs.includes(dogsMatch)) return false;
      }
    }
    
    // Good with kids filter - use _match field only, no text fallback
    // "unknown" option includes animals with unknown/empty/missing _match
    if (selectedGoodWithKids.length > 0) {
      const kidsMatch = animal.behaviorNotes?.goodWithKids_match;
      const isUnknown = !kidsMatch || kidsMatch === 'unknown';
      if (isUnknown) {
        if (!selectedGoodWithKids.includes('unknown')) return false;
      } else {
        if (!selectedGoodWithKids.includes(kidsMatch)) return false;
      }
    }
    
    // Special needs filter — OR within group
    if (selectedSpecialNeeds.length > 0) {
      const needsInfo = (animal.behaviorNotes?.specialNeeds || '').toLowerCase();
      const hasSpecialNeeds = needsInfo && needsInfo !== 'none' && needsInfo !== '-' && 
                              !needsInfo.includes('no special') && needsInfo !== 'not specified';
      const animalValue = hasSpecialNeeds ? 'yes' : 'no';
      if (!selectedSpecialNeeds.includes(animalValue)) return false;
    }
    
    // Search filter (color search - check both ShelterManager and behavior notes)
    if (search) {
      const smColor = (animal.color || '').toLowerCase();
      const bnColor = (animal.behaviorNotes?.color || '').toLowerCase();
      if (!smColor.includes(search) && !bnColor.includes(search)) {
        return false;
      }
    }
    
    return true;
  });
  
  // Calculate scores
  filteredAnimals = filteredAnimals.map(animal => ({
    ...animal,
    matchScore: calculateMatchScore(animal)
  }));
  
  // Sort: active filters → score desc; default → alphabetical by name
  const filtersActive = hasActiveFilters();
  if (filtersActive) {
    filteredAnimals.sort((a, b) => (b.matchScore || 0) - (a.matchScore || 0));
  } else {
    filteredAnimals.sort((a, b) => {
      const nameA = (a.name || '').toLowerCase();
      const nameB = (b.name || '').toLowerCase();
      return nameA.localeCompare(nameB);
    });
  }
  
  renderAnimals();
}

function showAnimalDetail(id) {
  const animal = filteredAnimals.find(a => a.id === id) || allAnimals.find(a => a.id === id);
  if (!animal) return;
  
  // Handle lead media: slot 2 video if available, else slot 1 photo
  const galleryLoading = document.querySelector('#modalGallery .gallery-loading');
  if (animal.video_url) {
    galleryLoading.innerHTML = `<video src="${animal.video_url}" class="modal-photo" autoplay loop muted playsinline onclick="event.preventDefault(); openLightbox('${animal.video_url}', 'video')"></video>`;
  } else {
    const img = document.createElement('img');
    img.id = 'modalPhoto';
    img.className = 'modal-photo';
    img.src = animal.photoUrl || PLACEHOLDER_DATA_URI;
    img.onerror = function() { this.src = PLACEHOLDER_DATA_URI; };
    galleryLoading.innerHTML = '';
    galleryLoading.appendChild(img);
  }
  
  // Populate modal details
  document.getElementById('modalName').textContent = animal.name;
  document.getElementById('modalGender').textContent = animal.sex;
  document.getElementById('modalAge').textContent = animal.age;
  document.getElementById('modalColor').textContent = animal.color;
  
  // Show score if available
  const scoreContainer = document.getElementById('modalScoreContainer');
  const scoreValue = document.getElementById('modalScore');
  if (animal.matchScore !== undefined) {
    scoreContainer.style.display = 'block';
    scoreValue.textContent = `${animal.matchScore}%`;
  } else {
    scoreContainer.style.display = 'none';
  }
  
  // Bio with language fallback chain
  const bioText = currentLang === 'es'
    ? (animal.bio_es_long || animal.bio_en_long || animal.description || '')
    : (animal.bio_en_long || animal.description || '');
  document.getElementById('modalDescription').textContent = bioText || i18n('modal.default_description');
  
  // Show modal
  elements.modal.classList.add('active');
  document.body.style.overflow = 'hidden';
  
  // Load photo gallery asynchronously
  loadModalGallery(animal);
}

function closeModal() {
  elements.modal.classList.remove('active');
  document.body.style.overflow = '';
  // Reset gallery to single photo placeholder
  document.getElementById('modalGallery').innerHTML = '<div class="gallery-loading"><img id="modalPhoto" src="" alt="" class="modal-photo"></div>';
}

// Gallery items for lightbox navigation (module scope) — {url, type: 'photo'|'video'}
let _galleryItems = [];

async function loadModalGallery(animal) {
  const galleryContainer = document.getElementById('modalGallery');
  
  try {
    const animalId = animal.animalId || animal.id;
    const response = await fetch(`${API_BASE}/photos/${animalId}`);
    const result = await response.json();
    
    if (!result.success || !result.data) return;
    
    const strip = result.data.strip || [];
    const publicPhotoUrl = animal.photoUrl || '';
    
    // Filter: strip positions 2+, photos only (skip videos), sorted by position
    const nonPublicPhotos = strip
      .filter(p => p.stripPosition > 1)
      .filter(p => p.mediaType !== 'video')
      .sort((a, b) => a.stripPosition - b.stripPosition);
    
    // Build thumbnail list: real photos only, up to 5. No logo padding.
    const photos = [];

    // When video is lead, slot 1 photo becomes the first thumbnail
    if (animal.video_url && publicPhotoUrl) {
      photos.push({ url: publicPhotoUrl, type: 'photo' });
    }

    for (let i = 0; i < 5 && i < nonPublicPhotos.length; i++) {
      photos.push({ url: nonPublicPhotos[i].photoUrl || nonPublicPhotos[i].fileUrl, type: 'photo' });
    }
    // Add public photo if not already included and we have room (non-video case)
    if (!animal.video_url && publicPhotoUrl && photos.length < 5 && !photos.some(p => p.url === publicPhotoUrl)) {
      photos.push({ url: publicPhotoUrl, type: 'photo' });
    }
    
    // Cap at 5
    photos.splice(5);
    
    // If no photos at all, keep showing the lead (video or placeholder)
    if (photos.length === 0) return;
    
    // Build lightbox rotation: video first (if present), then all photos
    if (animal.video_url) {
      _galleryItems = [{ url: animal.video_url, type: 'video' }, ...photos];
    } else {
      _galleryItems = photos.slice();
    }

    if (animal.video_url) {
      // Video lead: keep existing <video> in gallery-loading, add photo thumbnails below
      const existingLead = galleryContainer.querySelector('.gallery-loading');
      const leadHtml = existingLead ? existingLead.outerHTML : '';
      galleryContainer.innerHTML = `
        <div class="gallery-layout">
          <div class="gallery-main">${leadHtml}</div>
          <div class="gallery-grid">
            ${photos.map(p => `
              <div class="gallery-grid-item">
                <img src="${p.url}" alt="" class="gallery-img"
                     onclick="openLightbox('${p.url}', 'photo')">
              </div>
            `).join('')}
          </div>
        </div>
      `;
    } else {
      // Photo lead: first photo is main, rest are thumbnails
      const showThumbs = photos.length > 1;
      galleryContainer.innerHTML = `
        <div class="gallery-layout${showThumbs ? '' : ' gallery-single'}">
          <div class="gallery-main">
            <img src="${photos[0].url}" alt="" class="gallery-img"
                 onclick="openLightbox('${photos[0].url}', 'photo')">
          </div>
          ${showThumbs ? `
            <div class="gallery-grid">
              ${photos.slice(1).map(p => `
                <div class="gallery-grid-item">
                  <img src="${p.url}" alt="" class="gallery-img"
                       onclick="openLightbox('${p.url}', 'photo')">
                </div>
              `).join('')}
            </div>
          ` : ''}
        </div>
      `;
    }
  } catch (err) {
    console.error('Gallery load failed:', err);
  }
}

// ============ Lightbox with Arrow Navigation (photos + video) ============

let _lightboxItems = [];
let _lightboxIndex = 0;

function _renderLightboxMedia(item) {
  if (item.type === 'video') {
    return `<video id="lightboxNavMedia" src="${item.url}" controls autoplay loop muted playsinline
              style="max-width:90vw; max-height:90vh; display:block; border-radius:8px;"></video>`;
  }
  return `<img id="lightboxNavMedia" src="${item.url}" alt=""
            style="max-width:90vw; max-height:90vh; display:block; border-radius:8px;">`;
}

function openLightbox(url, type) {
  _lightboxItems = _galleryItems.slice();
  _lightboxIndex = _lightboxItems.findIndex(p => p.url === url);
  if (_lightboxIndex < 0) _lightboxIndex = 0;

  const item = _lightboxItems[_lightboxIndex];
  const showArrows = _lightboxItems.length > 1;
  const overlay = document.createElement('div');
  overlay.className = 'photo-lightbox';
  overlay.onclick = (e) => { if (e.target === overlay) overlay.remove(); };
  overlay.innerHTML = `
    <div class="lightbox-image-wrap" onclick="event.stopPropagation()">
      ${showArrows ? `
        <button class="lightbox-nav lightbox-nav-left" onclick="event.stopPropagation(); lightboxNav(-1)" aria-label="${i18n('lightbox.prev_aria')}">\u2039</button>
      ` : ''}
      ${_renderLightboxMedia(item)}
      ${showArrows ? `
        <button class="lightbox-nav lightbox-nav-right" onclick="event.stopPropagation(); lightboxNav(1)" aria-label="${i18n('lightbox.next_aria')}">\u203A</button>
      ` : ''}
    </div>
  `;
  document.body.appendChild(overlay);
}

// Backward compat alias for inline onclick handlers
function openPhotoLightbox(url) { openLightbox(url, 'photo'); }

function lightboxNav(delta) {
  if (_lightboxItems.length <= 1) return;
  _lightboxIndex = (_lightboxIndex + delta + _lightboxItems.length) % _lightboxItems.length;
  const item = _lightboxItems[_lightboxIndex];
  const old = document.getElementById('lightboxNavMedia');
  if (!old || !item) return;

  if (item.type === 'video' && old.tagName === 'VIDEO') {
    old.src = item.url;
  } else if (item.type === 'photo' && old.tagName === 'IMG') {
    old.src = item.url;
  } else {
    // Type changed — swap element
    const temp = document.createElement('div');
    temp.innerHTML = _renderLightboxMedia(item);
    old.replaceWith(temp.firstElementChild);
  }
}

  // Uncheck all checkboxes for all filter groups

      cb.checked = false;
    });

  // Clear search
  elements.searchInput.value = '';
  
  applyFilters();
}

// Utility: Debounce
function debounce(func, wait) {
  let timeout;
  return function executedFunction(...args) {
    const later = () => {
      clearTimeout(timeout);
      func(...args);
    };
    clearTimeout(timeout);
    timeout = setTimeout(later, wait);
  };
}
