function isGatedWrite(method: string, p: string): boolean {
  // --- Adoption applications ---
  // PATCH /api/adoption-applications/:id (born-gated W1a)
  if (method === 'PATCH' && /^\/api\/adoption-applications\/\d+$/.test(p)) return true;

  // --- Volunteer PII writes (W1b) ---
  // POST /api/volunteers/upload
  if (method === 'POST' && /^\/api\/volunteers\/upload$/.test(p)) return true;
  // POST /api/volunteers/rotate-image
  if (method === 'POST' && /^\/api\/volunteers\/rotate-image$/.test(p)) return true;
  // PATCH /api/volunteers/:id
  if (method === 'PATCH' && /^\/api\/volunteers\/\d+$/.test(p)) return true;
  // DELETE /api/volunteers/:id
  if (method === 'DELETE' && /^\/api\/volunteers\/\d+$/.test(p)) return true;
  // PATCH /api/volunteers/:id/policy-reviewed
  if (method === 'PATCH' && /^\/api\/volunteers\/\d+\/policy-reviewed$/.test(p)) return true;
  // PATCH /api/volunteers/:id/approval

  if (method === 'POST' && /^\/api\/volunteers\/\d+\/commitments$/.test(p)) return true;
  // DELETE /api/volunteers/:id/commitments
  if (method === 'DELETE' && /^\/api\/volunteers\/\d+\/commitments$/.test(p)) return true;
  // POST /api/volunteers/:id/declines
  if (method === 'POST' && /^\/api\/volunteers\/\d+\/declines$/.test(p)) return true;
  // DELETE /api/volunteers/:id/declines
  if (method === 'DELETE' && /^\/api\/volunteers\/\d+\/declines$/.test(p)) return true;
  // POST /api/volunteers/timeclock/manual
  if (method === 'POST' && /^\/api\/volunteers\/timeclock\/manual$/.test(p)) return true;

  // --- Intake writes (W1b) ---
  // POST /api/intakes/:id/status
  if (method === 'POST' && /^\/api\/intakes\/\d+\/status$/.test(p)) return true;
  // POST /api/dashboard/intake/:id/health-assessment
  if (method === 'POST' && /^\/api\/dashboard\/intake\/\d+\/health-assessment$/.test(p)) return true;
  // POST /api/dashboard/intake/:id/seizure-record
  if (method === 'POST' && /^\/api\/dashboard\/intake\/\d+\/seizure-record$/.test(p)) return true;

  if (method === 'POST' && /^\/api\/intake-recipients$/.test(p)) return true;
  // DELETE /api/intake-recipients/:id
  if (method === 'DELETE' && /^\/api\/intake-recipients\/\d+$/.test(p)) return true;

  return false;
}
